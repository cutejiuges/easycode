#!/usr/bin/env python3
from pathlib import Path
import datetime,json,resource,shutil,subprocess,sys,time
folder=Path('/Users/cutejiuge/.codex/worktrees/62a7/easycode/openspec/changes/add-runtime-effort-control/evidence/regression-2026-11-15')
argv=[shutil.which('go'),*sys.argv[1:]]
if len(argv)<2 or argv[1]!='test':
    raise SystemExit(subprocess.call(argv))
argv=[argv[0],'test','-json','-count=1',*argv[2:]]
name='test-race' if '-race' in argv else 'test'
started=datetime.datetime.now(datetime.timezone.utc).isoformat(); start=time.monotonic(); usage=resource.getrusage(resource.RUSAGE_CHILDREN)
meta={'command':subprocess.list2cmdline(argv),'started_at':started,'uncached':True}
(folder/(name+'-timing.json')).write_text(json.dumps(meta,indent=2)+'\n')
passed=[]; failures=[]; packages=[]
with (folder/(name+'-events.jsonl')).open('w') as output:
    process=subprocess.Popen(argv,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
    for line in process.stdout:
        output.write(line); output.flush()
        try: event=json.loads(line)
        except ValueError:
            print(line,end='',flush=True); continue
        if event.get('Action') in ('pass','skip','fail'):
            if event.get('Test'):
                (failures if event['Action']=='fail' else passed).append(event)
            else:
                packages.append(event)
                print(event.get('Package'),event['Action'],event.get('Elapsed',0),'seconds',flush=True)
        if event.get('Action')=='fail' and event.get('Test'): print('FAILED:',event.get('Test'),flush=True)
    result=process.wait()
after=resource.getrusage(resource.RUSAGE_CHILDREN)
roots=[e for e in passed+failures if '/' not in e['Test']]
meta.update(exit_code=result,elapsed_seconds=round(time.monotonic()-start,3),child_cpu_user_seconds=round(after.ru_utime-usage.ru_utime,3),child_cpu_system_seconds=round(after.ru_stime-usage.ru_stime,3),package_results=packages,root_results=roots,passed_or_skipped_tests=len(passed),failed_tests=failures)
(folder/(name+'-timing.json')).write_text(json.dumps(meta,indent=2)+'\n')
raise SystemExit(result)
